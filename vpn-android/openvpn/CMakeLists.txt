# Pin full commits, including mbed TLS's framework submodule.
function(fetch_source name repository revision)
    set(cache "${CMAKE_CURRENT_SOURCE_DIR}/../build/openvpn-sources")
    file(MAKE_DIRECTORY "${cache}")
    file(LOCK "${cache}/${name}.lock" GUARD FUNCTION TIMEOUT 300)
    set(source "${cache}/${name}-${revision}")
    if(EXISTS "${source}/.git")
        execute_process(COMMAND git -C "${source}" rev-parse HEAD OUTPUT_VARIABLE head OUTPUT_STRIP_TRAILING_WHITESPACE)
        if(head STREQUAL revision)
            if(name STREQUAL "mbedtls" AND NOT EXISTS "${source}/framework/CMakeLists.txt")
                execute_process(COMMAND git -C "${source}" submodule update --init -- framework COMMAND_ERROR_IS_FATAL ANY)
            endif()
            set(${name}_SOURCE_DIR "${source}" PARENT_SCOPE)
            return()
        endif()
        message(FATAL_ERROR "Invalid cached revision for ${name}")
    endif()
    FetchContent_Declare(${name} SOURCE_DIR "${source}" GIT_REPOSITORY https://github.com/${repository}.git GIT_TAG ${revision})
    FetchContent_GetProperties(${name})
    if(NOT ${name}_POPULATED)
        FetchContent_Populate(${name})
    endif()
    set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
endfunction()

# OpenVPN verifies the profile's CA, server certificate role and optional
# verify-x509-name rather than treating a VPN remote IP as an HTTPS hostname.
# mbed TLS 3.6 requires that the VPN client select this PKI policy explicitly.
function(adapt_core_header header old replacement)
    file(READ "${ovpn_SOURCE_DIR}/${header}" source)
    string(FIND "${source}" "${old}" match)
    if(match EQUAL -1)
        message(FATAL_ERROR "Recheck ${header} against the pinned mbed TLS release")
    endif()
    string(REPLACE "${old}" "${replacement}" source "${source}")
    get_filename_component(dir "${header}" DIRECTORY)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/overlay/${dir}")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/overlay/${header}" CONTENT "${source}" @ONLY)
endfunction()
adapt_core_header("openvpn/mbedtls/ssl/sslctx.hpp"
    "if (hostname && ((c.flags & SSLConst::ENABLE_CLIENT_SNI) || c.ca_chain))"
    "if ((hostname && (c.flags & SSLConst::ENABLE_CLIENT_SNI)) || c.ca_chain)")
# mbed TLS 3.6 requires an explicit padding selection before CBC encrypt/decrypt.
adapt_core_header("openvpn/mbedtls/crypto/cipher.hpp" "// set key and encrypt/decrypt mode"
    "if (mbedtls_cipher_info_get_mode(ci) == MBEDTLS_MODE_CBC && mbedtls_cipher_set_padding_mode(&ctx, MBEDTLS_PADDING_PKCS7) != 0) throw mbedtls_cipher_error(\"CBC padding\");")
